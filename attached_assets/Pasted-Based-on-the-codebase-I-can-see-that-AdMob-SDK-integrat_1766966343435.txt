package com.lingualink.speechtranslator
import android.app.Activity
import android.util.Log
import android.webkit.JavascriptInterface
import android.webkit.WebView
import com.google.android.gms.ads.*
import com.google.android.gms.ads.interstitial.InterstitialAd
import com.google.android.gms.ads.interstitial.InterstitialAdLoadCallback
import com.google.android.gms.ads.rewarded.RewardedAd
import com.google.android.gms.ads.rewarded.RewardedAdLoadCallback
class AdBridge(
    private val activity: Activity,
    private val webView: WebView
) {
    private var interstitialAd: InterstitialAd? = null
    private var rewardedAd: RewardedAd? = null
    
    companion object {
        private const val INTERSTITIAL_AD_UNIT = "ca-app-pub-9277938970928959/1473642031"
        private const val REWARDED_AD_UNIT = "ca-app-pub-9277938970928959/8777416980"
    }
    
    init {
        MobileAds.initialize(activity) {
            Log.d("AdBridge", "AdMob initialized")
            loadInterstitialAd()
            loadRewardedAd()
        }
    }
    
    private fun loadInterstitialAd() {
        InterstitialAd.load(
            activity,
            INTERSTITIAL_AD_UNIT,
            AdRequest.Builder().build(),
            object : InterstitialAdLoadCallback() {
                override fun onAdLoaded(ad: InterstitialAd) {
                    interstitialAd = ad
                    notifyWeb("interstitialLoaded", "true")
                }
                
                override fun onAdFailedToLoad(error: LoadAdError) {
                    Log.e("AdBridge", "Interstitial failed: ${error.message}")
                    interstitialAd = null
                }
            }
        )
    }
    
    private fun loadRewardedAd() {
        RewardedAd.load(
            activity,
            REWARDED_AD_UNIT,
            AdRequest.Builder().build(),
            object : RewardedAdLoadCallback() {
                override fun onAdLoaded(ad: RewardedAd) {
                    rewardedAd = ad
                    notifyWeb("rewardedLoaded", "true")
                }
                
                override fun onAdFailedToLoad(error: LoadAdError) {
                    Log.e("AdBridge", "Rewarded failed: ${error.message}")
                    rewardedAd = null
                }
            }
        )
    }
    
    @JavascriptInterface
    fun showInterstitial() {
        activity.runOnUiThread {
            interstitialAd?.let { ad ->
                ad.fullScreenContentCallback = object : FullScreenContentCallback() {
                    override fun onAdDismissedFullScreenContent() {
                        notifyWeb("interstitialClosed", "")
                        loadInterstitialAd()
                    }
                    
                    override fun onAdShowedFullScreenContent() {
                        notifyWeb("interstitialShown", "")
                        interstitialAd = null
                    }
                }
                ad.show(activity)
            } ?: run {
                notifyWeb("interstitialFailed", "Ad not ready")
            }
        }
    }
    
    @JavascriptInterface
    fun showRewarded() {
        activity.runOnUiThread {
            rewardedAd?.show(activity) { rewardItem ->
                notifyWeb("rewardedEarned", rewardItem.amount.toString())
            } ?: run {
                notifyWeb("rewardedFailed", "Ad not ready")
            }
        }
    }
    
    @JavascriptInterface
    fun isInterstitialReady(): Boolean = interstitialAd != null
    
    @JavascriptInterface
    fun isRewardedAdReady(): Boolean = rewardedAd != null
    
    private fun notifyWeb(event: String, data: String) {
        val js = "window.onAdBridgeEvent('$event', '$data');"
        activity.runOnUiThread {
            webView.evaluateJavascript(js, null)
        }
    }
}
2. Register in MainActivity.kt

class MainActivity : AppCompatActivity() {
    private lateinit var webView: WebView
    private lateinit var adBridge: AdBridge
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        webView = WebView(this).apply {
            settings.javaScriptEnabled = true
            settings.domStorageEnabled = true
        }
        
        // CRITICAL: Register bridge BEFORE loading URL
        adBridge = AdBridge(this, webView)
        webView.addJavascriptInterface(adBridge, "AdBridge")
        
        webView.loadUrl("https://your-repl.replit.app")
        setContentView(webView)
    }
}
