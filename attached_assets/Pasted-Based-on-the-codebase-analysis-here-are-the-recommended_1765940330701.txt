2. Implement Lifecycle-Aware Ad Display

// Add to AdBridge.kt or create new LifecycleAwareAdManager.kt
class LifecycleAwareAdManager(
    private val activity: AppCompatActivity
) : DefaultLifecycleObserver {
    
    private var isInForeground = true
    private var pendingAdShow: (() -> Unit)? = null
    
    init {
        activity.lifecycle.addObserver(this)
    }
    
    override fun onResume(owner: LifecycleOwner) {
        isInForeground = true
        // Show pending ad if any
        pendingAdShow?.invoke()
        pendingAdShow = null
    }
    
    override fun onPause(owner: LifecycleOwner) {
        isInForeground = false
    }
    
    // Wrap ad show in lifecycle check
    fun showInterstitial(ad: InterstitialAd, callback: (Boolean) -> Unit) {
        if (!isInForeground) {
            Log.w("AdMob", "Activity not in foreground, queuing ad")
            pendingAdShow = { showInterstitial(ad, callback) }
            return
        }
        
        activity.lifecycleScope.launch(Dispatchers.Main) {
            if (activity.lifecycle.currentState.isAtLeast(Lifecycle.State.RESUMED)) {
                ad.show(activity)
                callback(true)
            } else {
                Log.w("AdMob", "Activity not resumed, skipping ad")
                callback(false)
            }
        }
    }
}
3. Add Impression Tracking Callbacks

private fun setupInterstitialCallbacks() {
    interstitialAd?.fullScreenContentCallback = object : FullScreenContentCallback() {
        override fun onAdShowedFullScreenContent() {
            Log.d("AdMob", "✅ Interstitial SHOWING - waiting for impression")
        }
        
        override fun onAdImpression() {
            Log.d("AdMob", "💰 IMPRESSION COUNTED - This is the key event!")
            // Notify WebView that impression was counted
            notifyWebView("interstitialImpression", "true")
        }
        
        override fun onAdDismissedFullScreenContent() {
            Log.d("AdMob", "Interstitial dismissed")
            resolveCallback(pendingInterstitialCallback, true)
            pendingInterstitialCallback = null
            interstitialAd = null
            loadInterstitialAd()
        }
        
        override fun onAdFailedToShowFullScreenContent(error: AdError) {
            Log.e("AdMob", "Interstitial failed: ${error.message}")
            resolveCallback(pendingInterstitialCallback, false, error.message)
            pendingInterstitialCallback = null
            interstitialAd = null
            loadInterstitialAd()
        }
    }
}

class AdPreloader(private val context: Context) {
    
    private var interstitialAd: InterstitialAd? = null
    private val retryDelays = listOf(1000L, 2000L, 4000L, 8000L, 16000L)
    private var retryCount = 0
    
    fun preloadInterstitial() {
        val adRequest = AdRequest.Builder().build()
        
        InterstitialAd.load(
            context,
            "ca-app-pub-9277938970928959/1473642031",
            adRequest,
            object : InterstitialAdLoadCallback() {
                override fun onAdLoaded(ad: InterstitialAd) {
                    interstitialAd = ad
                    retryCount = 0
                    Log.d("AdMob", "✅ Interstitial preloaded and ready")
                }
                
                override fun onAdFailedToLoad(error: LoadAdError) {
                    Log.e("AdMob", "❌ Load failed: ${error.message}")
                    scheduleRetry()
                }
            }
        )
    }
    
    private fun scheduleRetry() {
        if (retryCount < retryDelays.size) {
            val delay = retryDelays[retryCount]
            retryCount++
            Handler(Looper.getMainLooper()).postDelayed({
                preloadInterstitial()
            }, delay)
        }
    }
}
