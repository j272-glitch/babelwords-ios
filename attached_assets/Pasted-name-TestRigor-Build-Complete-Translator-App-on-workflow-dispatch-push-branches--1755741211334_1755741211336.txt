          package com.lingualink.translator.activity
          
          import android.Manifest
          import android.content.Intent
          import android.content.pm.PackageManager
          import android.os.Bundle
          import android.util.Log
          import android.view.Menu
          import android.view.MenuItem
          import android.widget.*
          import androidx.appcompat.app.AppCompatActivity
          import androidx.core.app.ActivityCompat
          import androidx.core.content.ContextCompat
          import com.google.android.gms.common.ConnectionResult
          import com.google.android.gms.common.GoogleApiAvailability
          import com.lingualink.translator.R
          import com.lingualink.translator.service.TranslationService
          import com.lingualink.translator.utils.LanguageUtils
          import com.lingualink.translator.utils.PreferencesManager
          
          class MainActivity : AppCompatActivity() {
              
              companion object {
                  private const val TAG = "LinguaLinkMain"
                  private const val PERMISSIONS_REQUEST_CODE = 1001
                  private const val CAMERA_REQUEST_CODE = 1002
                  private const val WEBVIEW_REQUEST_CODE = 1003
              }
              
              private lateinit var sourceLanguageSpinner: Spinner
              private lateinit var targetLanguageSpinner: Spinner
              private lateinit var inputEditText: EditText
              private lateinit var outputTextView: TextView
              private lateinit var translateButton: Button
              private lateinit var swapLanguagesButton: ImageButton
              private lateinit var cameraButton: Button
              private lateinit var webviewButton: Button
              private lateinit var statusTextView: TextView
              
              private lateinit var preferencesManager: PreferencesManager
              private lateinit var translationService: TranslationService
              
              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  
                  try {
                      setContentView(R.layout.activity_main)
                      
                      initializeComponents()
                      checkPermissions()
                      checkPlayServices()
                      setupLanguageSpinners()
                      setupClickListeners()
                      handleIncomingIntent()
                      
                      Log.d(TAG, "MainActivity created successfully")
                      updateStatus("Ready for translation")
                      
                  } catch (e: Exception) {
                      Log.e(TAG, "Error in onCreate", e)
                      showErrorMessage("App initialization failed: ${e.message}")
                  }
              }
              
              private fun initializeComponents() {
                  sourceLanguageSpinner = findViewById(R.id.sourceLanguageSpinner)
                  targetLanguageSpinner = findViewById(R.id.targetLanguageSpinner)
                  inputEditText = findViewById(R.id.inputEditText)
                  outputTextView = findViewById(R.id.outputTextView)
                  translateButton = findViewById(R.id.translateButton)
                  swapLanguagesButton = findViewById(R.id.swapLanguagesButton)
                  cameraButton = findViewById(R.id.cameraButton)
                  webviewButton = findViewById(R.id.webviewButton)
                  statusTextView = findViewById(R.id.statusTextView)
                  
                  preferencesManager = PreferencesManager(this)
                  translationService = TranslationService(this)
              }
              
              private fun checkPermissions() {
                  val permissions = arrayOf(
                      // REQUIRED WEBVIEW PERMISSIONS
                      Manifest.permission.RECORD_AUDIO,           // For WebView speech recognition
                      Manifest.permission.MODIFY_AUDIO_SETTINGS,  // For WebView microphone access
                      // Note: INTERNET permission is automatically granted
                      
                      // ADDITIONAL APP PERMISSIONS
                      Manifest.permission.CAMERA,                 // For OCR features
                      Manifest.permission.ACCESS_FINE_LOCATION    // For location-aware translation
                  )
                  
                  val permissionsToRequest = permissions.filter {
                      ContextCompat.checkSelfPermission(this, it) != PackageManager.PERMISSION_GRANTED
                  }
                  
                  if (permissionsToRequest.isNotEmpty()) {
                      Log.d(TAG, "Requesting permissions: ${permissionsToRequest.joinToString()}")
                      ActivityCompat.requestPermissions(this, permissionsToRequest.toTypedArray(), PERMISSIONS_REQUEST_CODE)
                  } else {
                      Log.d(TAG, "All required permissions already granted")
                      updateStatus("All permissions granted - WebView ready")
                  }
              }
              
              private fun checkPlayServices(): Boolean {
                  val apiAvailability = GoogleApiAvailability.getInstance()
                  val resultCode = apiAvailability.isGooglePlayServicesAvailable(this)
                  
                  return when (resultCode) {
                      ConnectionResult.SUCCESS -> {
                          Log.d(TAG, "Google Play Services is available")
                          updateStatus("Google Play Services available")
                          true
                      }
                      else -> {
                          Log.w(TAG, "Google Play Services not available: $resultCode")
                          updateStatus("Using offline translation mode")
                          false
                      }
                  }
              }
              
              private fun setupLanguageSpinners() {
                  val languages = LanguageUtils.getSupportedLanguages()
                  val adapter = ArrayAdapter(this, android.R.layout.simple_spinner_item, languages.map { it.displayName })
                  adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item)
                  
                  sourceLanguageSpinner.adapter = adapter
                  targetLanguageSpinner.adapter = adapter
                  
                  // Set default languages
                  sourceLanguageSpinner.setSelection(languages.indexOfFirst { it.code == "en" })
                  targetLanguageSpinner.setSelection(languages.indexOfFirst { it.code == "es" })
              }
              
              private fun setupClickListeners() {
                  translateButton.setOnClickListener {
                      performTranslation()
                  }
                  
                  swapLanguagesButton.setOnClickListener {
                      swapLanguages()
                  }
                  
                  cameraButton.setOnClickListener {
                      openCamera()
                  }
                  
                  webviewButton.setOnClickListener {
                      openWebView()
                  }
              }
              
              private fun performTranslation() {
                  val inputText = inputEditText.text.toString().trim()
                  if (inputText.isEmpty()) {
                      showErrorMessage("Please enter text to translate")
                      return
                  }
                  
                  updateStatus("Translating...")
                  translateButton.isEnabled = false
                  
                  val sourceLanguage = LanguageUtils.getSupportedLanguages()[sourceLanguageSpinner.selectedItemPosition]
                  val targetLanguage = LanguageUtils.getSupportedLanguages()[targetLanguageSpinner.selectedItemPosition]
                  
                  translationService.translate(inputText, sourceLanguage.code, targetLanguage.code) { result ->
                      runOnUiThread {
                          translateButton.isEnabled = true
                          when {
                              result.isSuccess -> {
                                  outputTextView.text = result.getOrNull()
                                  updateStatus("Translation completed")
                              }
                              else -> {
                                  showErrorMessage("Translation failed: ${result.exceptionOrNull()?.message}")
                                  updateStatus("Translation failed")
                              }
                          }
                      }
                  }
              }
              
              private fun swapLanguages() {
                  val sourcePosition = sourceLanguageSpinner.selectedItemPosition
                  val targetPosition = targetLanguageSpinner.selectedItemPosition
                  
                  sourceLanguageSpinner.setSelection(targetPosition)
                  targetLanguageSpinner.setSelection(sourcePosition)
                  
                  // Also swap the text content
                  val inputText = inputEditText.text.toString()
                  val outputText = outputTextView.text.toString()
                  
                  if (outputText.isNotEmpty()) {
                      inputEditText.setText(outputText)
                      outputTextView.text = inputText
                  }
              }
              
              private fun openCamera() {
                  if (ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA) == PackageManager.PERMISSION_GRANTED) {
                      val intent = Intent(this, CameraActivity::class.java)
                      startActivityForResult(intent, CAMERA_REQUEST_CODE)
                  } else {
                      showErrorMessage("Camera permission required for OCR")
                  }
              }
              
              private fun openWebView() {
                  val intent = Intent(this, WebViewActivity::class.java)
                  // Pass the GTLingua custom domain URL to WebView
                  intent.putExtra("url", "https://gtlingua.com")
                  startActivityForResult(intent, WEBVIEW_REQUEST_CODE)
              }
              
              private fun handleIncomingIntent() {
                  val intent = intent
                  if (Intent.ACTION_SEND == intent.action && "text/plain" == intent.type) {
                      val sharedText = intent.getStringExtra(Intent.EXTRA_TEXT)
                      if (!sharedText.isNullOrEmpty()) {
                          inputEditText.setText(sharedText)
                          updateStatus("Text received from external app")
                      }
                  }
              }
              
              private fun updateStatus(message: String) {
                  statusTextView.text = message
                  Log.d(TAG, "Status: $message")
              }
              
              private fun showErrorMessage(message: String) {
                  Toast.makeText(this, message, Toast.LENGTH_LONG).show()
                  Log.e(TAG, message)
              }
              
              override fun onCreateOptionsMenu(menu: Menu): Boolean {
                  menuInflater.inflate(R.menu.main_menu, menu)
                  return true
              }
              
              override fun onOptionsItemSelected(item: MenuItem): Boolean {
                  return when (item.itemId) {
                      R.id.action_settings -> {
                          // Open settings
                          true
                      }
                      R.id.action_history -> {
                          // Open translation history
                          true
                      }
                      else -> super.onOptionsItemSelected(item)
                  }
              }
              
              override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
                  super.onActivityResult(requestCode, resultCode, data)
                  
                  when (requestCode) {
                      CAMERA_REQUEST_CODE -> {
                          if (resultCode == RESULT_OK && data != null) {
                              val extractedText = data.getStringExtra("extracted_text")
                              if (!extractedText.isNullOrEmpty()) {
                                  inputEditText.setText(extractedText)
                                  updateStatus("Text extracted from image")
                              }
                          }
                      }
                      WEBVIEW_REQUEST_CODE -> {
                          if (resultCode == RESULT_OK && data != null) {
                              val translatedText = data.getStringExtra("translated_text")
                              if (!translatedText.isNullOrEmpty()) {
                                  outputTextView.text = translatedText
                                  updateStatus("Translation received from web service")
                              }
                          }
                      }
                  }
              }
              
              override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
                  super.onRequestPermissionsResult(requestCode, permissions, grantResults)
                  
                  if (requestCode == PERMISSIONS_REQUEST_CODE) {
                      val grantedPermissions = mutableListOf<String>()
                      val deniedPermissions = mutableListOf<String>()
                      
                      permissions.forEachIndexed { index, permission ->
                          if (grantResults[index] == PackageManager.PERMISSION_GRANTED) {
                              grantedPermissions.add(permission)
                          } else {
                              deniedPermissions.add(permission)
                          }
                      }
                      
                      Log.d(TAG, "Granted permissions: ${grantedPermissions.joinToString()}")
                      Log.d(TAG, "Denied permissions: ${deniedPermissions.joinToString()}")
                      
                      // Check specifically for WebView required permissions
                      val webViewPermissionsGranted = grantedPermissions.any { it == Manifest.permission.RECORD_AUDIO } &&
                                                     grantedPermissions.any { it == Manifest.permission.MODIFY_AUDIO_SETTINGS }
                      
                      when {
                          grantResults.all { it == PackageManager.PERMISSION_GRANTED } -> {
                              updateStatus("All permissions granted - Full functionality available")
                          }
                          webViewPermissionsGranted -> {
                              updateStatus("WebView permissions granted - Speech features available")
                          }
                          deniedPermissions.contains(Manifest.permission.RECORD_AUDIO) -> {
                              updateStatus("Audio permission denied - Voice features limited")
                              showErrorMessage("Voice translation features require microphone access")
                          }
                          else -> {
                              updateStatus("Some permissions denied - Limited functionality")
                          }
                      }
                  }
              }
          }
          EOF
          
          echo "✅ MainActivity.kt created with full functionality"

          import android.content.Intent
          import android.graphics.Bitmap
          import android.os.Bundle
          import android.util.Log

          import android.view.View
          import android.webkit.*
          import android.widget.ProgressBar
          import androidx.appcompat.app.AppCompatActivity
          import androidx.appcompat.widget.Toolbar
          import com.lingualink.translator.R

          class WebViewActivity : AppCompatActivity() {
              
              companion object {
                  private const val TAG = "WebViewActivity"
                  // Use the GTLingua custom domain or fallback to Google Translate

              }
              
              private lateinit var webView: WebView
              private lateinit var progressBar: ProgressBar
              private lateinit var toolbar: Toolbar
              
              @SuppressLint("SetJavaScriptEnabled")
              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  setContentView(R.layout.activity_webview)
                  
                  initializeViews()
                  setupToolbar()
                  setupWebView()
                  
                  // Priority: Intent URL > Default Replit URL > Fallback
                  val url = intent.getStringExtra("url") ?: DEFAULT_URL
                  loadUrl(url)
              }
              
              private fun initializeViews() {
                  webView = findViewById(R.id.webView)
                  progressBar = findViewById(R.id.progressBar)
                  toolbar = findViewById(R.id.toolbar)
              }
              
              private fun setupToolbar() {
                  setSupportActionBar(toolbar)
                  supportActionBar?.apply {
                      setDisplayHomeAsUpEnabled(true)
                      setDisplayShowHomeEnabled(true)
                      title = "LinguaLink Web"
                  }
              }
              
              @SuppressLint("SetJavaScriptEnabled")
              private fun setupWebView() {
                  val webSettings = webView.settings
                  
                  // Key WebView settings for optimal functionality
                  webSettings.javaScriptEnabled = true
                  webSettings.mediaPlaybackRequiresUserGesture = false
                  webSettings.domStorageEnabled = true
                  
                  // Additional essential settings
                  webSettings.databaseEnabled = true
                  webSettings.cacheMode = WebSettings.LOAD_DEFAULT
                  
                  // Enable zoom controls
                  webSettings.setSupportZoom(true)
                  webSettings.builtInZoomControls = true
                  webSettings.displayZoomControls = false
                  
                  // Enhanced user agent for GTLingua compatibility
                  webSettings.userAgentString = "Mozilla/5.0 (Linux; Android 11; Mobile LinguaLink) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.120 Mobile Safari/537.36"
                  
                  // Security settings (balanced for functionality)
                  webSettings.allowFileAccess = false
                  webSettings.allowContentAccess = false
                  webSettings.allowFileAccessFromFileURLs = false
                  webSettings.allowUniversalAccessFromFileURLs = false
                  
                  // Mixed content mode for HTTPS custom domain
                  webSettings.mixedContentMode = WebSettings.MIXED_CONTENT_COMPATIBILITY_MODE
                  
                  // Additional settings for modern web apps and media
                  webSettings.loadWithOverviewMode = true
                  webSettings.useWideViewPort = true
