# Android Container App: Complete AdMob Impressions Troubleshooting Guide

## The Core Problem

You’re seeing **100% match rate but zero impressions**. This means AdMob is successfully finding ads to serve, but they’re never actually rendering on screen. This guide addresses every layer where this can fail in a WebView-based hybrid app architecture.

-----

### Verify Correct SDK Versions


### Check Google Play Services Availability

```kotlin
// Call this before any ad operations
fun checkPlayServices(context: Context): Boolean {
    val googleApiAvailability = GoogleApiAvailability.getInstance()
    val resultCode = googleApiAvailability.isGooglePlayServicesAvailable(context)
    
    if (resultCode != ConnectionResult.SUCCESS) {
        Log.e("AdMob", "Google Play Services unavailable: $resultCode")
        if (googleApiAvailability.isUserResolvableError(resultCode)) {
            googleApiAvailability.getErrorDialog(context as Activity, resultCode, 9000)?.show()
        }
        return false
    }
    return true
}
```

### Common Issue: Play Services Version Mismatch

Many devices in India run outdated Play Services. Handle gracefully:

```kotlin
// In your Application class
override fun onCreate() {

    // Initialize Mobile Ads SDK with callback

            if (status.initializationState != AdapterStatus.State.READY) {
                Log.w("AdMob", "Adapter not ready: ${status.description}")

        }
```

### WebView-Native Ad Bridge

The key issue with WebView apps is that ads load in native code but must coordinate with your web content:

```kotlin
class AdBridge(private val activity: Activity, private val webView: WebView) {
    
    private var interstitialAd: InterstitialAd? = null
    private var rewardedAd: RewardedAd? = null
    private var isAdReady = false
    
    // JavaScript interface for web app communication
    @JavascriptInterface
    fun loadInterstitial() {
        activity.runOnUiThread {
            loadInterstitialAd()
        }
    }
    
    @JavascriptInterface
    fun showInterstitial(): Boolean {
        var shown = false
        activity.runOnUiThread {
            interstitialAd?.let { ad ->
                ad.show(activity)
                shown = true
            } ?: run {
                Log.w("AdBridge", "Interstitial not ready - loading now")
                loadInterstitialAd()
            }
        }
        return shown
    }
    
    @JavascriptInterface
    fun isInterstitialReady(): Boolean = isAdReady
    
    private fun loadInterstitialAd() {
        val adRequest = AdRequest.Builder().build()
        
        InterstitialAd.load(
            activity,
            "ca-app-pub-XXXX/YYYY", // Your ad unit ID
            adRequest,
            object : InterstitialAdLoadCallback() {
                override fun onAdLoaded(ad: InterstitialAd) {
                    interstitialAd = ad
                    isAdReady = true
                    
                    // Notify web app
                    notifyWebApp("onInterstitialLoaded", "true")
                    
                    // Set up show callback
                    ad.fullScreenContentCallback = object : FullScreenContentCallback() {
                        override fun onAdShowedFullScreenContent() {
                            Log.d("AdBridge", "Ad showed - IMPRESSION COUNTED")
                            notifyWebApp("onInterstitialImpression", "true")
                        }
                        
                        override fun onAdDismissedFullScreenContent() {
                            isAdReady = false
                            interstitialAd = null
                            notifyWebApp("onInterstitialClosed", "true")
                            // Preload next ad
                            loadInterstitialAd()
                        }
                        
                        override fun onAdFailedToShowFullScreenContent(error: AdError) {
                            Log.e("AdBridge", "Failed to show: ${error.message}")
                            isAdReady = false
                            interstitialAd = null
                            notifyWebApp("onInterstitialShowFailed", error.message)
                        }
                    }
                }
                
                override fun onAdFailedToLoad(error: LoadAdError) {
                    Log.e("AdBridge", "Failed to load: ${error.message}")
                    isAdReady = false
                    notifyWebApp("onInterstitialLoadFailed", error.message)
                }
            }
        )
    }
    
    private fun notifyWebApp(event: String, data: String) {
        activity.runOnUiThread {
            webView.evaluateJavascript(
                "window.dispatchEvent(new CustomEvent('$event', {detail: '$data'}))",
                null
            )
        }
    }
}
```

### Web App Integration (TypeScript/React side)

```typescript
// adBridge.ts
declare global {
  interface Window {
    NativeAdBridge?: {
      loadInterstitial: () => void;
      showInterstitial: () => boolean;
      isInterstitialReady: () => boolean;
    };
  }
}

export class AdManager {
  private static instance: AdManager;
  
  static getInstance(): AdManager {
    if (!AdManager.instance) {
      AdManager.instance = new AdManager();
    }
    return AdManager.instance;
  }
  
  constructor() {
    // Listen for native ad events
    window.addEventListener('onInterstitialLoaded', () => {
      console.log('Interstitial ready');
    });
    
    window.addEventListener('onInterstitialImpression', () => {
      console.log('IMPRESSION RECORDED');
      // Track in your analytics
    });
    
    window.addEventListener('onInterstitialShowFailed', (e: CustomEvent) => {
      console.error('Show failed:', e.detail);
    });
  }
  
  loadInterstitial(): void {
    if (window.NativeAdBridge) {
      window.NativeAdBridge.loadInterstitial();
    }
  }
  
  showInterstitial(): boolean {
    if (window.NativeAdBridge?.isInterstitialReady()) {
      return window.NativeAdBridge.showInterstitial();
    }
    return false;
  }
  
  isReady(): boolean {
    return window.NativeAdBridge?.isInterstitialReady() ?? false;
  }
}
```

## 4. Activity Lifecycle Management

### The #1 Cause of “Match but No Impression”

If your Activity is finishing or in the background when `show()` is called, the ad will not display:

```kotlin
class MainActivity : AppCompatActivity() {
    
    private var interstitialAd: InterstitialAd? = null
    private var isActivityActive = false
    
    override fun onResume() {
        super.onResume()
        isActivityActive = true
    }
    
    override fun onPause() {
        super.onPause()
        isActivityActive = false
    }
    
    override fun onDestroy() {
        super.onDestroy()
        interstitialAd = null
    }
    
    fun showInterstitialSafely() {
        // CRITICAL: Check activity state before showing
        if (!isActivityActive) {
            Log.w("Ads", "Activity not active - cannot show ad")
            return
        }
        
        if (isFinishing || isDestroyed) {
            Log.w("Ads", "Activity finishing/destroyed - cannot show ad")
            return
        }
        
        interstitialAd?.show(this) ?: run {
            Log.w("Ads", "Ad not loaded yet")
        }
    }
}
```

## 5. Banner Ad Implementation

### Common Banner Issues in WebView Apps

```kotlin
class BannerAdManager(
    private val activity: Activity,
    private val bannerContainer: FrameLayout
) {
    private var adView: AdView? = null
    
    fun loadBanner() {
        // Remove any existing banner
        adView?.destroy()
        bannerContainer.removeAllViews()
        
        adView = AdView(activity).apply {
            // CRITICAL: Set ad size before loading
            setAdSize(AdSize.BANNER)
            adUnitId = "ca-app-pub-XXXX/YYYY"
            
            // Must add to container BEFORE loading
            bannerContainer.addView(this)
            
            adListener = object : AdListener() {
                override fun onAdLoaded() {
                    Log.d("Banner", "Loaded successfully")
                    
                    // CRITICAL: Verify the banner is actually visible
                    post {
                        if (visibility == View.VISIBLE && 
                            width > 0 && height > 0 &&
                            isShown) {
                            Log.d("Banner", "Banner is visible on screen")
                        } else {
                            Log.e("Banner", "Banner loaded but NOT visible!")
                            Log.e("Banner", "visibility=$visibility, w=$width, h=$height, shown=$isShown")
                        }
                    }
                }
                
                override fun onAdImpression() {
                    Log.d("Banner", "IMPRESSION RECORDED")
                }
                
                override fun onAdFailedToLoad(error: LoadAdError) {
                    Log.e("Banner", "Failed: ${error.code} - ${error.message}")
                }
            }
            
            loadAd(AdRequest.Builder().build())
        }
    }
    
    fun verifyBannerVisibility() {
        adView?.let { banner ->
            val location = IntArray(2)
            banner.getLocationOnScreen(location)
            
            val screenHeight = activity.resources.displayMetrics.heightPixels
            val screenWidth = activity.resources.displayMetrics.widthPixels
            
            val isOnScreen = location[0] >= 0 && 
                             location[1] >= 0 &&
                             location[0] + banner.width <= screenWidth &&
                             location[1] + banner.height <= screenHeight
            
            Log.d("Banner", "Position: (${location[0]}, ${location[1]})")
            Log.d("Banner", "Size: ${banner.width} x ${banner.height}")
            Log.d("Banner", "Is on screen: $isOnScreen")
        }
    }
}
```

## 6. Rewarded Ad Implementation

```kotlin
class RewardedAdManager(private val activity: Activity) {
    
    private var rewardedAd: RewardedAd? = null
    private var isLoading = false
    
    fun loadRewardedAd() {
        if (isLoading) return
        isLoading = true
        
        val adRequest = AdRequest.Builder().build()
        
        RewardedAd.load(
            activity,
            "ca-app-pub-XXXX/YYYY",
            adRequest,
            object : RewardedAdLoadCallback() {
                override fun onAdLoaded(ad: RewardedAd) {
                    isLoading = false
                    rewardedAd = ad
                    setupFullScreenCallback(ad)
                    Log.d("Rewarded", "Ad loaded successfully")
                }
                
                override fun onAdFailedToLoad(error: LoadAdError) {
                    isLoading = false
                    rewardedAd = null
                    Log.e("Rewarded", "Failed to load: ${error.code} - ${error.message}")
                    
                    // Retry with exponential backoff
                    Handler(Looper.getMainLooper()).postDelayed({
                        loadRewardedAd()
                    }, 30000) // 30 second delay before retry
                }
            }
        )
    }
    
    private fun setupFullScreenCallback(ad: RewardedAd) {
        ad.fullScreenContentCallback = object : FullScreenContentCallback() {
            override fun onAdShowedFullScreenContent() {
                Log.d("Rewarded", "IMPRESSION - Ad showed")
            }
            
            override fun onAdDismissedFullScreenContent() {
                rewardedAd = null
                loadRewardedAd() // Preload next
            }
            
            override fun onAdFailedToShowFullScreenContent(error: AdError) {
                Log.e("Rewarded", "Failed to show: ${error.code} - ${error.message}")
                rewardedAd = null
                loadRewardedAd()
            }
        }
    }
    
    fun showRewardedAd(onRewarded: (RewardItem) -> Unit): Boolean {
        val ad = rewardedAd
        
        if (ad == null) {
            Log.w("Rewarded", "Ad not ready")
            loadRewardedAd()
            return false
        }
        
        // Activity state check
        if (activity.isFinishing || activity.isDestroyed) {
            Log.w("Rewarded", "Activity not valid")
            return false
        }
        
        ad.show(activity) { rewardItem ->
            Log.d("Rewarded", "User earned reward: ${rewardItem.amount} ${rewardItem.type}")
            onRewarded(rewardItem)
        }
        
        return true
    }
}
```

## 10. India-Specific Optimizations

### Geo-Based Ad Configuration

```kotlin
class GeoAwareAdManager(private val context: Context) {
    
    fun getOptimalAdRequest(): AdRequest {
        val builder = AdRequest.Builder()
        
        // Add content targeting for better India fill rates
        val extras = Bundle().apply {
            // Request family-safe ads (higher fill in India)

        // Priority order for India market based on fill rates
        return listOf(
            "rewarded",      // Best fill + eCPM
            "interstitial",  // Good fill
            "native",        // Moderate fill
            "banner"         // Lowest eCPM but consistent
        )
    }
}
```

### Preload Strategy for Low-Bandwidth

```kotlin
class AdPreloader(private val context: Context) {
    
    private val adCache = mutableMapOf<String, Any>()
    private val maxCacheAge = 30 * 60 * 1000L // 30 minutes
    
    fun preloadAdsOnWifi() {
        val cm = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
        val caps = cm.getNetworkCapabilities(cm.activeNetwork)
        
        if (caps?.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) == true) {
            // Preload multiple ad types when on WiFi
            preloadInterstitial()
            preloadRewarded()
        } else {
            // On mobile data, only preload rewarded (highest value)
            preloadRewarded()
        }
    }
}
```

## Quick Diagnostic for Zero Impressions

Run through this in order:

1. **Is the ad actually loading?** Check for `onAdLoaded` callback
1. **Is `show()` being called?** Add logging before the call
1. **Is Activity valid when `show()` is called?** Check `isFinishing`, `isDestroyed`
1. **Is the ad view visible?** Use Layout Inspector
1. **Are there any `onAdFailedToShow` errors?** Check logcat
1. **Is this a timing issue?** Add delays, verify lifecycle

If all checks pass and you still see 100% match / 0% impressions, the issue is likely in the WebView-native bridge communication. Add extensive logging to every callback and trace the flow.
