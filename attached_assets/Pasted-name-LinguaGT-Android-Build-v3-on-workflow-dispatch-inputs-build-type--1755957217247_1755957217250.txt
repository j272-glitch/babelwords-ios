          #!/bin/sh
          
          ##############################################################################
          ##
          ##  Gradle start up script for UN*X
          ##
          ##############################################################################
          
          # Attempt to set APP_HOME
          PRG="$0"
          while [ -h "$PRG" ] ; do
              ls=`ls -ld "$PRG"`
              link=`expr "$ls" : '.*-> \(.*\)$'`
              if expr "$link" : '/.*' > /dev/null; then
                  PRG="$link"
              else
                  PRG=`dirname "$PRG"`"/$link"
              fi
          done
          SAVED="`pwd`"
          cd "`dirname \"$PRG\"`/" >/dev/null
          APP_HOME="`pwd -P`"
          cd "$SAVED" >/dev/null
          
          APP_NAME="Gradle"
          APP_BASE_NAME=`basename "$0"`
          DEFAULT_JVM_OPTS='"-Xmx64m" "-Xms64m"'
          MAX_FD="maximum"
          
          warn () {
              echo "$*"
          }
          
          die () {
              echo
              echo "$*"
              echo
              exit 1
          }
          
          # OS specific support
          cygwin=false
          msys=false
          darwin=false
          nonstop=false
          case "`uname`" in
            CYGWIN* )
              cygwin=true
              ;;
            Darwin* )
              darwin=true
              ;;
            MINGW* )
              msys=false
              ;;
            NONSTOP* )
              nonstop=true
              ;;
          esac
          
          CLASSPATH=$APP_HOME/gradle/wrapper/gradle-wrapper.jar
          
          # Determine the Java command to use to start the JVM
          if [ -n "$JAVA_HOME" ] ; then
              if [ -x "$JAVA_HOME/jre/sh/java" ] ; then
                  JAVACMD="$JAVA_HOME/jre/sh/java"
              else
                  JAVACMD="$JAVA_HOME/bin/java"
              fi
              if [ ! -x "$JAVACMD" ] ; then
                  die "ERROR: JAVA_HOME is set to an invalid directory: $JAVA_HOME"
              fi
          else
              JAVACMD="java"
              which java >/dev/null 2>&1 || die "ERROR: JAVA_HOME is not set and no 'java' command could be found in your PATH."
          fi
          
          # Increase the maximum file descriptors if we can
          if [ "$cygwin" = "false" -a "$darwin" = "false" -a "$nonstop" = "false" ] ; then
              MAX_FD_LIMIT=`ulimit -H -n`
              if [ $? -eq 0 ] ; then
                  if [ "$MAX_FD" = "maximum" -o "$MAX_FD" = "max" ] ; then
                      MAX_FD="$MAX_FD_LIMIT"
                  fi
                  ulimit -n $MAX_FD
                  if [ $? -ne 0 ] ; then
                      warn "Could not set maximum file descriptor limit: $MAX_FD"
                  fi
              else
                  warn "Could not query maximum file descriptor limit: $MAX_FD_LIMIT"
              fi
          fi
          
          # For Darwin, add options to specify how the application appears in the dock
          if $darwin; then
              GRADLE_OPTS="$GRADLE_OPTS \"-Xdock:name=$APP_NAME\" \"-Xdock:icon=$APP_HOME/media/gradle.icns\""
          fi
          
          # For Cygwin or MSYS, switch paths to Windows format before running java
          if [ "$cygwin" = "true" -o "$msys" = "true" ] ; then
              APP_HOME=`cygpath --path --mixed "$APP_HOME"`
              CLASSPATH=`cygpath --path --mixed "$CLASSPATH"`
              JAVACMD=`cygpath --unix "$JAVACMD"`
          fi
          
          # Collect all arguments for the java command
          eval set -- $DEFAULT_JVM_OPTS $JAVA_OPTS $GRADLE_OPTS "\"-Dorg.gradle.appname=$APP_BASE_NAME\"" -classpath "\"$CLASSPATH\"" org.gradle.wrapper.GradleWrapperMain "$@"
          
          exec "$JAVACMD" "$@"
          GRADLEW
          
          chmod +x gradlew

              else
                  echo "- Signing: Debug keystore ⚠️" >> build-report.md
              fi
              echo "" >> build-report.md
          fi
          
          if [ -f "app/build/outputs/bundle/release/app-release.aab" ]; then
              echo "### ✅ Release AAB (App Bundle)" >> build-report.md
              echo "- Size: $(du -h app/build/outputs/bundle/release/app-release.aab | cut -f1)" >> build-report.md
              echo "- Path: \`app/build/outputs/bundle/release/app-release.aab\`" >> build-report.md

                  echo "- Status: Ready for Google Play Store upload ✅" >> build-report.md
              else
                  echo "- Status: Test only (debug signed) ⚠️" >> build-report.md
              fi
              echo "" >> build-report.md
          fi
          
          # Add deployment instructions with proper quoting
          cat >> build-report.md << 'INSTRUCTIONS_EOF'
          
          ## Deployment Instructions
          
          ### For Testing (APK)
          1. Download the APK artifact from this workflow run
          2. Enable "Install from unknown sources" on your Android device
          3. Transfer and install the APK file
          
          ### For Google Play Store (AAB)
          INSTRUCTIONS_EOF
          
          if [ "$HAS_VALID_RELEASE_KEYSTORE" = "true" ]; then
              cat >> build-report.md << 'PROD_EOF'
          1. Download the AAB artifact from this workflow run
          2. Go to [Google Play Console](https://play.google.com/console)
          3. Navigate to your app > Release > Production
          4. Upload the AAB file
          5. Complete the release process
          PROD_EOF
          else
              cat >> build-report.md << 'DEBUG_EOF'
          ⚠️ **Note:** The current AAB is signed with a debug key and cannot be uploaded to Google Play Store.
          
          To enable production releases, add these secrets to your repository:
          - `ANDROID_KEYSTORE_BASE64` - Base64 encoded keystore file
          - `ANDROID_KEYSTORE_PASSWORD` - Keystore password
          - `ANDROID_KEY_ALIAS` - Key alias name
          - `ANDROID_KEY_PASSWORD` - Key password
          DEBUG_EOF
          fi
          
          # Display report
          echo ""
          echo "================================"
          cat build-report.md
          echo "================================"
