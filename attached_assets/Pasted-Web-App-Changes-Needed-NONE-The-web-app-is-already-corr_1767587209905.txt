✅ Registers window.onRewardEarned callback

Kotlin Changes Needed: ALL
The Kotlin AdMobBridge.kt must be completely rewritten to:

1. Add Parameter-less Methods
@JavascriptInterface
fun showInterstitial() {  // NO placementId param
    activity.runOnUiThread { showInterstitialAd() }
}
@JavascriptInterface
fun showRewarded() {  // NO placementId param
    activity.runOnUiThread { showRewardedAd() }
}
@JavascriptInterface
fun isInterstitialReady(): Boolean = interstitialAd != null
@JavascriptInterface
fun isRewardedReady(): Boolean = rewardedAd != null
2. Add Auto-Preload on Bridge Creation
init {
    initializeAdMob()
    // Auto-preload after SDK init
    preloadInterstitial()
    preloadRewarded()
}
3. Add Callback Injection to WebView
private fun notifyAdReady(adType: String) {
    activity.runOnUiThread {
        webView.evaluateJavascript(
            "if(window.onNativeAdReady) window.onNativeAdReady('$adType')", 
            null
        )
    }
}
private fun notifyRewardEarned(type: String, amount: Int) {
    activity.runOnUiThread {
        webView.evaluateJavascript(
            "if(window.onRewardEarned) window.onRewardEarned('$type', $amount)", 
            null
        )
    }
}
4. Call Callbacks on Ad Load Success
override fun onAdLoaded(ad: InterstitialAd) {
    interstitialAd = ad
    notifyAdReady("interstitial")  // ← NEW
}
override fun onUserEarnedReward(reward: RewardItem) {
    notifyRewardEarned(reward.type, reward.amount)  // ← NEW
}
5. Auto-Reload After Ad Dismissed
override fun onAdDismissedFullScreenContent() {
    interstitialAd = null
    preloadInterstitial()  // Immediately preload next
}
