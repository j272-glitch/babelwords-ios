      - name: Setup Android SDK
        run: |
          echo "=== SETTING UP ANDROID SDK ==="
          
          # Use Android SDK from GitHub Actions environment if available
          if [ -d "$ANDROID_SDK_ROOT" ]; then
              echo "Using existing Android SDK at $ANDROID_SDK_ROOT"
              export ANDROID_HOME=$ANDROID_SDK_ROOT
          else
              # Install Android SDK command line tools
              SDK_TOOLS_URL="https://dl.google.com/android/repository/commandlinetools-linux-11076708_latest.zip"
              wget -q $SDK_TOOLS_URL -O android-sdk-tools.zip
              
              # Create Android SDK directory
              export ANDROID_HOME=$HOME/android-sdk
              mkdir -p $ANDROID_HOME/cmdline-tools
              
              # Extract command line tools
              unzip -q android-sdk-tools.zip -d $ANDROID_HOME/cmdline-tools
              mv $ANDROID_HOME/cmdline-tools/cmdline-tools $ANDROID_HOME/cmdline-tools/latest
          fi
          
          # Add to PATH
          echo "$ANDROID_HOME/cmdline-tools/latest/bin" >> $GITHUB_PATH
          echo "$ANDROID_HOME/platform-tools" >> $GITHUB_PATH
          echo "ANDROID_HOME=$ANDROID_HOME" >> $GITHUB_ENV
          echo "ANDROID_SDK_ROOT=$ANDROID_HOME" >> $GITHUB_ENV
          
          # Fix Minor Issue #1: Suppress broken pipe error
          # Accept licenses silently without broken pipe error
          echo "Accepting Android SDK licenses..."
          yes 2>/dev/null | $ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager --licenses > /dev/null 2>&1 || true
          
          # Install required packages with retry logic
          echo "Installing required SDK packages..."
          for i in 1 2 3; do
              $ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager \
                "platform-tools" \
                "platforms;android-34" \
                "build-tools;34.0.0" && break

          # Create gradlew script
          cat > gradlew << 'GRADLEW'
          #!/bin/sh
          
          ##############################################################################
          ##
          ##  Gradle start up script for UN*X
          ##
          ##############################################################################
          
          # Attempt to set APP_HOME
          PRG="$0"
          while [ -h "$PRG" ] ; do
              ls=`ls -ld "$PRG"`
              link=`expr "$ls" : '.*-> \(.*\)$'`
              if expr "$link" : '/.*' > /dev/null; then
                  PRG="$link"
              else
                  PRG=`dirname "$PRG"`"/$link"
              fi
          done
          SAVED="`pwd`"
          cd "`dirname \"$PRG\"`/" >/dev/null
          APP_HOME="`pwd -P`"
          cd "$SAVED" >/dev/null
          
          APP_NAME="Gradle"
          APP_BASE_NAME=`basename "$0"`
          DEFAULT_JVM_OPTS='"-Xmx64m" "-Xms64m"'
          MAX_FD="maximum"
          
          warn () {
              echo "$*"
          }
          
          die () {
              echo
              echo "$*"
              echo
              exit 1
          }
          
          # OS specific support
          cygwin=false
          msys=false
          darwin=false
          nonstop=false
          case "`uname`" in
            CYGWIN* )
              cygwin=true
              ;;
            Darwin* )
              darwin=true
              ;;
            MINGW* )
              msys=false
              ;;
            NONSTOP* )
              nonstop=true
              ;;
          esac
          
          CLASSPATH=$APP_HOME/gradle/wrapper/gradle-wrapper.jar
          
          # Determine the Java command to use to start the JVM
          if [ -n "$JAVA_HOME" ] ; then
              if [ -x "$JAVA_HOME/jre/sh/java" ] ; then
                  JAVACMD="$JAVA_HOME/jre/sh/java"
              else
                  JAVACMD="$JAVA_HOME/bin/java"
              fi
              if [ ! -x "$JAVACMD" ] ; then
                  die "ERROR: JAVA_HOME is set to an invalid directory: $JAVA_HOME"
              fi
          else
              JAVACMD="java"
              which java >/dev/null 2>&1 || die "ERROR: JAVA_HOME is not set and no 'java' command could be found in your PATH."
          fi
          
          # Increase the maximum file descriptors if we can
          if [ "$cygwin" = "false" -a "$darwin" = "false" -a "$nonstop" = "false" ] ; then
              MAX_FD_LIMIT=`ulimit -H -n`
              if [ $? -eq 0 ] ; then
                  if [ "$MAX_FD" = "maximum" -o "$MAX_FD" = "max" ] ; then
                      MAX_FD="$MAX_FD_LIMIT"
                  fi
                  ulimit -n $MAX_FD
                  if [ $? -ne 0 ] ; then
                      warn "Could not set maximum file descriptor limit: $MAX_FD"
                  fi
              else
                  warn "Could not query maximum file descriptor limit: $MAX_FD_LIMIT"
              fi
          fi
          
          # For Darwin, add options to specify how the application appears in the dock
          if $darwin; then
              GRADLE_OPTS="$GRADLE_OPTS \"-Xdock:name=$APP_NAME\" \"-Xdock:icon=$APP_HOME/media/gradle.icns\""
          fi
          
          # For Cygwin or MSYS, switch paths to Windows format before running java
          if [ "$cygwin" = "true" -o "$msys" = "true" ] ; then
              APP_HOME=`cygpath --path --mixed "$APP_HOME"`
              CLASSPATH=`cygpath --path --mixed "$CLASSPATH"`
              JAVACMD=`cygpath --unix "$JAVACMD"`
          fi
          
          # Collect all arguments for the java command
          eval set -- $DEFAULT_JVM_OPTS $JAVA_OPTS $GRADLE_OPTS "\"-Dorg.gradle.appname=$APP_BASE_NAME\"" -classpath "\"$CLASSPATH\"" org.gradle.wrapper.GradleWrapperMain "$@"
          
          exec "$JAVACMD" "$@"
          GRADLEW
          
          chmod +x gradlew
          echo "✅ Gradle wrapper setup complete"

          package com.lingualink.linguagt;
          
          import android.Manifest;
          import android.content.pm.PackageManager;
          import android.os.Build;
          import android.os.Bundle;
          import android.util.Log;

          import androidx.appcompat.app.AppCompatActivity;
          import androidx.core.app.ActivityCompat;

          public class MainActivity extends AppCompatActivity {
              private static final String TAG = "LinguaGT";
              private WebView webView;
              private static final int PERMISSION_REQUEST_CODE = 100;
              
              @Override
              protected void onCreate(Bundle savedInstanceState) {
                  super.onCreate(savedInstanceState);
                  
                  // Request permissions
                  requestPermissions();
                  
                  // Setup WebView
                  webView = new WebView(this);
                  setContentView(webView);
                  
                  setupWebView();
              }
              
              private void setupWebView() {
                  WebSettings webSettings = webView.getSettings();
                  
                  // Enable JavaScript and other required settings
                  webSettings.setJavaScriptEnabled(true);
                  webSettings.setDomStorageEnabled(true);
                  webSettings.setDatabaseEnabled(true);
                  webSettings.setMediaPlaybackRequiresUserGesture(false);
                  webSettings.setAllowFileAccess(true);
                  webSettings.setAllowContentAccess(true);
                  webSettings.setLoadsImagesAutomatically(true);
                  webSettings.setMixedContentMode(WebSettings.MIXED_CONTENT_COMPATIBILITY_MODE);
                  
                  // Enhanced settings for LinguaGT
                  webSettings.setJavaScriptCanOpenWindowsAutomatically(true);
                  webSettings.setCacheMode(WebSettings.LOAD_DEFAULT);
                  webSettings.setUserAgentString("Mozilla/5.0 (Linux; Android 11; LinguaGT) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Mobile Safari/537.36");
                  
                  // Support for modern web features
                  webSettings.setGeolocationEnabled(true);
                  
                  // WebView client
                  webView.setWebViewClient(new WebViewClient() {
                      @Override
                      public boolean shouldOverrideUrlLoading(WebView view, String url) {
                          Log.d(TAG, "Loading URL: " + url);
                          return false;
                      }
                      
                      @Override
                      public void onPageFinished(WebView view, String url) {
                          super.onPageFinished(view, url);
                          Log.d(TAG, "Page loaded: " + url);
                      }
                  });
                  
                  // WebChrome client for permissions
                  webView.setWebChromeClient(new WebChromeClient() {
                      @Override
                      public void onPermissionRequest(final PermissionRequest request) {
                          Log.d(TAG, "Permission requested: " + String.join(", ", request.getResources()));
                          runOnUiThread(new Runnable() {
                              @Override
                              public void run() {
                                  String[] resources = request.getResources();
                                  request.grant(resources);
                              }
                          });
                      }
                      
                      @Override
                      public boolean onConsoleMessage(ConsoleMessage consoleMessage) {
                          Log.d(TAG, "Console: " + consoleMessage.message());
                          return true;
                      }
                  });
                  
                  // Load LinguaGT
                  webView.loadUrl("https://linguagt.com");
              }
              
              private void requestPermissions() {
                  String[] permissions = {
                      Manifest.permission.RECORD_AUDIO,
                      Manifest.permission.CAMERA,
                      Manifest.permission.MODIFY_AUDIO_SETTINGS
                  };
                  
                  boolean needsPermission = false;
                  for (String permission : permissions) {
                      if (ContextCompat.checkSelfPermission(this, permission) != PackageManager.PERMISSION_GRANTED) {
                          needsPermission = true;
                          break;
                      }
                  }
                  
                  if (needsPermission) {
                      ActivityCompat.requestPermissions(this, permissions, PERMISSION_REQUEST_CODE);
                  }
              }
              
              @Override
              public void onRequestPermissionsResult(int requestCode, String[] permissions, int[] grantResults) {
                  super.onRequestPermissionsResult(requestCode, permissions, grantResults);
                  if (requestCode == PERMISSION_REQUEST_CODE) {
                      boolean allGranted = true;
                      for (int result : grantResults) {
                          if (result != PackageManager.PERMISSION_GRANTED) {
                              allGranted = false;
                              break;
                          }
                      }
                      if (!allGranted) {
                          Toast.makeText(this, "Some features may not work without permissions", Toast.LENGTH_LONG).show();
                      }
                  }
              }
              
              @Override
              public void onBackPressed() {
                  if (webView.canGoBack()) {
                      webView.goBack();
                  } else {
                      super.onBackPressed();
                  }
              }
              
              @Override
              protected void onDestroy() {
                  if (webView != null) {
                      webView.destroy();
                  }
                  super.onDestroy();
              }
          }

      - name: Create Launcher Icons
        run: |
          echo "=== CREATING LAUNCHER ICONS ==="
          
          # Install ImageMagick
          sudo apt-get update && sudo apt-get install -y imagemagick
          
          # Create launcher icons with better design
          for size in 48 72 96 144 192; do
              density=""
              case $size in
                  48) density="mdpi" ;;
                  72) density="hdpi" ;;
                  96) density="xhdpi" ;;
                  144) density="xxhdpi" ;;
                  192) density="xxxhdpi" ;;
              esac
              
              # Create a gradient background with text
              convert -size ${size}x${size} \
                      xc: -sparse-color Barycentric \
                      "0,0 #4F46E5 ${size},${size} #7C3AED" \
                      -fill white \
                      -gravity center \
                      -pointsize $((size/3)) \
                      -font "DejaVu-Sans-Bold" \
                      -annotate +0+0 "LG" \
                      "app/src/main/res/mipmap-${density}/ic_launcher.png"
          done
          
          echo "✅ Icons created"

      - name: Clean and Prepare Build
        run: |
          echo "=== CLEANING BUILD DIRECTORIES ==="
          
          # Make gradlew executable
          chmod +x gradlew
          
          # Clean any existing build artifacts
          ./gradlew clean --stacktrace
          
          # Remove any intermediate directories that might cause issues
          rm -rf app/build/intermediates 2>/dev/null || true
          
          echo "✅ Build directories cleaned"

      - name: Build Debug APK
        if: env.BUILD_TYPE == 'all' || env.BUILD_TYPE == 'debug'
        run: |
          echo "=== BUILDING DEBUG APK ==="
          
          # Display build environment
          echo "Java version: $(java -version 2>&1 | head -n 1)"
          echo "Gradle version: $(./gradlew --version | grep Gradle | cut -d ' ' -f 2)"
          
          # Build debug APK with progress
          ./gradlew assembleDebug --stacktrace --console=plain
          
          # Verify debug APK
          if [ -f app/build/outputs/apk/debug/app-debug.apk ]; then
              echo "✅ Debug APK built successfully"
              echo "Size: $(du -h app/build/outputs/apk/debug/app-debug.apk | cut -f1)"
              
              # Extract APK info
              aapt dump badging app/build/outputs/apk/debug/app-debug.apk | grep -E "package:|versionCode|versionName" || true
          else
              echo "❌ Debug APK build failed"
              exit 1
          fi

      - name: Build Release APK
        if: env.BUILD_TYPE == 'all' || env.BUILD_TYPE == 'release'
        run: |
          echo "=== BUILDING RELEASE APK ==="
          
          # Build release APK with progress
          ./gradlew assembleRelease --stacktrace --console=plain
          
          # Verify release APK
          if [ -f app/build/outputs/apk/release/app-release.apk ]; then
              echo "✅ Release APK built successfully"
              echo "Size: $(du -h app/build/outputs/apk/release/app-release.apk | cut -f1)"
              
              # Verify signature
              jarsigner -verify -verbose -certs app/build/outputs/apk/release/app-release.apk | head -20 || true
              
              # Extract APK info
              aapt dump badging app/build/outputs/apk/release/app-release.apk | grep -E "package:|versionCode|versionName" || true
          else
              echo "❌ Release APK build failed"
              exit 1
          fi

      - name: Build Release AAB
        if: env.BUILD_TYPE == 'all' || env.BUILD_TYPE == 'release'
        run: |
          echo "=== BUILDING RELEASE AAB ==="
          
          # Build release AAB with progress
          ./gradlew bundleRelease --stacktrace --console=plain
          
          # Verify release AAB
          if [ -f app/build/outputs/bundle/release/app-release.aab ]; then
              echo "✅ Release AAB built successfully"
              echo "Size: $(du -h app/build/outputs/bundle/release/app-release.aab | cut -f1)"
              
              # Verify signature
              jarsigner -verify app/build/outputs/bundle/release/app-release.aab && echo "✅ AAB is signed" || echo "⚠️ AAB signature verification failed"
          else
              echo "❌ Release AAB build failed"
              exit 1
          fi

      - name: Set Build Outputs
        id: build_outputs
        if: always()
        run: |
          # Set output paths for artifacts
          if [ -f "app/build/outputs/apk/debug/app-debug.apk" ]; then
              echo "debug_apk=app/build/outputs/apk/debug/app-debug.apk" >> $GITHUB_OUTPUT
          fi
          
          if [ -f "app/build/outputs/apk/release/app-release.apk" ]; then
              echo "release_apk=app/build/outputs/apk/release/app-release.apk" >> $GITHUB_OUTPUT
          fi
          
          if [ -f "app/build/outputs/bundle/release/app-release.aab" ]; then
              echo "release_aab=app/build/outputs/bundle/release/app-release.aab" >> $GITHUB_OUTPUT
          fi
          
          # Set build status
          if [ -f "app/build/outputs/apk/debug/app-debug.apk" ] || [ -f "app/build/outputs/apk/release/app-release.apk" ] || [ -f "app/build/outputs/bundle/release/app-release.aab" ]; then
              echo "status=success" >> $GITHUB_OUTPUT
          else
              echo "status=failed" >> $GITHUB_OUTPUT
          fi

      - name: Upload Debug APK
        uses: actions/upload-artifact@v4
        if: (env.BUILD_TYPE == 'all' || env.BUILD_TYPE == 'debug') && always()
        with:
          name: linguagt-debug-apk-v${{ env.VERSION_NAME }}
          path: app/build/outputs/apk/debug/app-debug.apk
          retention-days: 30
          if-no-files-found: warn

      - name: Upload Release APK
        uses: actions/upload-artifact@v4
        if: (env.BUILD_TYPE == 'all' || env.BUILD_TYPE == 'release') && always()
        with:
          name: linguagt-release-apk-v${{ env.VERSION_NAME }}
          path: app/build/outputs/apk/release/app-release.apk
          retention-days: 90
          if-no-files-found: warn

      - name: Upload Release AAB
        uses: actions/upload-artifact@v4
        if: (env.BUILD_TYPE == 'all' || env.BUILD_TYPE == 'release') && always()
        with:
          name: linguagt-release-aab-v${{ env.VERSION_NAME }}
          path: app/build/outputs/bundle/release/app-release.aab
          retention-days: 90
          if-no-files-found: warn

      - name: Generate Build Report
        if: always()
        run: |
          # Fix Minor Issue #2: Use unquoted EOF to allow command expansion
          # Generate timestamp
          BUILD_DATE=$(date -u +"%Y-%m-%d %H:%M:%S UTC")
          
          cat > build-report.md << EOF
          # LinguaGT Android Build Report
          
          ## Build Information
          - **Repository:** ${{ github.repository }}
          - **Branch:** ${{ github.ref_name }}
          - **Commit:** ${{ github.sha }}
          - **Run ID:** ${{ github.run_id }}
          - **Build Type:** ${{ env.BUILD_TYPE }}
          - **Version:** ${{ env.VERSION_NAME }} (Code: ${{ env.VERSION_CODE }})
          - **Date:** ${BUILD_DATE}
          - **Runner:** ${{ runner.os }} ${{ runner.arch }}
          
          ## Build Artifacts
          EOF
          
          # Add artifact information
          if [ -f "app/build/outputs/apk/debug/app-debug.apk" ]; then
              echo "### ✅ Debug APK" >> build-report.md
              echo "- Size: $(du -h app/build/outputs/apk/debug/app-debug.apk | cut -f1)" >> build-report.md
              echo "- Path: \`app/build/outputs/apk/debug/app-debug.apk\`" >> build-report.md
              echo "" >> build-report.md
          fi
          
          if [ -f "app/build/outputs/apk/release/app-release.apk" ]; then
              echo "### ✅ Release APK" >> build-report.md
              echo "- Size: $(du -h app/build/outputs/apk/release/app-release.apk | cut -f1)" >> build-report.md
              echo "- Path: \`app/build/outputs/apk/release/app-release.apk\`" >> build-report.md
              if [ "$HAS_VALID_RELEASE_KEYSTORE" = "true" ]; then
                  echo "- Signing: Production keystore ✅" >> build-report.md
              else
                  echo "- Signing: Debug keystore ⚠️" >> build-report.md
              fi
              echo "" >> build-report.md
          fi
          
          if [ -f "app/build/outputs/bundle/release/app-release.aab" ]; then
              echo "### ✅ Release AAB (App Bundle)" >> build-report.md
              echo "- Size: $(du -h app/build/outputs/bundle/release/app-release.aab | cut -f1)" >> build-report.md
              echo "- Path: \`app/build/outputs/bundle/release/app-release.aab\`" >> build-report.md
              if [ "$HAS_VALID_RELEASE_KEYSTORE" = "true" ]; then
                  echo "- Status: Ready for Google Play Store upload ✅" >> build-report.md
              else
                  echo "- Status: Test only (debug signed) ⚠️" >> build-report.md
              fi
              echo "" >> build-report.md
          fi
          
          # Add deployment instructions with proper quoting
          cat >> build-report.md << 'INSTRUCTIONS_EOF'
          
          ## Deployment Instructions
          
          ### For Testing (APK)
          1. Download the APK artifact from this workflow run
          2. Enable "Install from unknown sources" on your Android device
          3. Transfer and install the APK file
          
          ### For Google Play Store (AAB)
          INSTRUCTIONS_EOF
          
          if [ "$HAS_VALID_RELEASE_KEYSTORE" = "true" ]; then
              cat >> build-report.md << 'PROD_EOF'
          1. Download the AAB artifact from this workflow run
          2. Go to [Google Play Console](https://play.google.com/console)
          3. Navigate to your app > Release > Production
          4. Upload the AAB file
          5. Complete the release process
          PROD_EOF
          else
              cat >> build-report.md << 'DEBUG_EOF'
          ⚠️ **Note:** The current AAB is signed with a debug key and cannot be uploaded to Google Play Store.
          
          To enable production releases, add these secrets to your repository:
          - `ANDROID_KEYSTORE_BASE64` - Base64 encoded keystore file
          - `ANDROID_KEYSTORE_PASSWORD` - Keystore password
          - `ANDROID_KEY_ALIAS` - Key alias name
          - `ANDROID_KEY_PASSWORD` - Key password
          DEBUG_EOF
          fi
          
          # Display report
          echo ""
          echo "================================"
          cat build-report.md
          echo "================================"

      - name: Upload Build Report
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: build-report-${{ github.run_id }}

          retention-days: 30

      - name: Create Release
        if: github.event_name == 'push' && github.ref == 'refs/heads/main' && steps.build_outputs.outputs.status == 'success' && env.CREATE_RELEASE == 'true'
        uses: softprops/action-gh-release@v1
        with:
          tag_name: v${{ env.VERSION_NAME }}
          name: LinguaGT v${{ env.VERSION_NAME }}
          draft: false
          prerelease: false
          generate_release_notes: true
          fail_on_unmatched_files: false
          files: |
            app/build/outputs/apk/debug/app-debug.apk
            app/build/outputs/apk/release/app-release.apk
            app/build/outputs/bundle/release/app-release.aab
            build-report.md
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        continue-on-error: true
      
      - name: Handle Release Creation Result
        if: github.event_name == 'push' && github.ref == 'refs/heads/main' && steps.build_outputs.outputs.status == 'success' && env.CREATE_RELEASE == 'true'
        run: |
          # Check if release was created successfully
          if gh release view "v${{ env.VERSION_NAME }}" > /dev/null 2>&1; then
              echo "✅ Release v${{ env.VERSION_NAME }} created successfully!"
              echo "View release at: https://github.com/${{ github.repository }}/releases/tag/v${{ env.VERSION_NAME }}"
          else
              echo "⚠️ Release creation may have failed. Possible reasons:"
              echo "  - Tag v${{ env.VERSION_NAME }} may already exist"
              echo "  - Insufficient permissions (check workflow permissions)"
              echo "  - Rate limiting or API issues"
              echo ""
              echo "You can manually create the release at:"
              echo "https://github.com/${{ github.repository }}/releases/new"
              echo ""
              echo "Artifacts are still available as workflow artifacts."
          fi
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
