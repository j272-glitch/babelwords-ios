          package com.lingualink.gtlingua.activity
          
          import android.Manifest
          import android.content.Intent
          import android.content.pm.PackageManager
          import android.os.Bundle
          import android.util.Log
          import android.widget.*
          import androidx.appcompat.app.AppCompatActivity
          import androidx.core.app.ActivityCompat
          import androidx.core.content.ContextCompat
          import com.lingualink.gtlingua.R
          import com.lingualink.gtlingua.utils.PreferencesManager
          
          class LauncherActivity : AppCompatActivity() {
              
              companion object {
                  private const val TAG = "GTLinguaLauncher"
                  private const val PERMISSIONS_REQUEST_CODE = 1001
                  private const val GTLINGUA_REQUEST_CODE = 2001
                  private const val CAMERA_REQUEST_CODE = 3001
              }
              
              // Clean, focused UI for GTLingua integration
              private lateinit var quickTextInput: EditText
              private lateinit var voiceTranslateButton: Button
              private lateinit var textTranslateButton: Button
              private lateinit var cameraOCRButton: Button
              private lateinit var statusText: TextView
              private lateinit var recentTranslations: TextView
              
              private lateinit var preferencesManager: PreferencesManager
              
              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  setContentView(R.layout.activity_launcher)
                  
                  try {
                      initializeComponents()
                      checkDeepgramPermissions()
                      setupClickListeners()
                      handleIncomingIntent()
                      
                      Log.d(TAG, "GTLingua Launcher ready")
                      updateStatus("Ready for 36-language translation with Deepgram AI")
                      
                  } catch (e: Exception) {
                      Log.e(TAG, "Error in onCreate", e)
                      Toast.makeText(this, "Initialization failed: ${e.message}", Toast.LENGTH_LONG).show()
                  }
              }
              
              private fun initializeComponents() {
                  quickTextInput = findViewById(R.id.quickTextInput)
                  voiceTranslateButton = findViewById(R.id.voiceTranslateButton)
                  textTranslateButton = findViewById(R.id.textTranslateButton)
                  cameraOCRButton = findViewById(R.id.cameraOCRButton)
                  statusText = findViewById(R.id.statusText)
                  recentTranslations = findViewById(R.id.recentTranslations)
                  
                  preferencesManager = PreferencesManager(this)
                  
                  // Load recent translations
                  loadRecentTranslations()
              }
              
              private fun checkDeepgramPermissions() {
                  val criticalPermissions = arrayOf(
                      Manifest.permission.RECORD_AUDIO,           // ESSENTIAL for Deepgram
                      Manifest.permission.MODIFY_AUDIO_SETTINGS   // ESSENTIAL for WebView audio
                  )
                  
                  val optionalPermissions = arrayOf(
                      Manifest.permission.CAMERA,                 // For OCR
                      Manifest.permission.ACCESS_FINE_LOCATION    // For context
                  )
                  
                  val allPermissions = criticalPermissions + optionalPermissions
                  val permissionsToRequest = allPermissions.filter {
                      ContextCompat.checkSelfPermission(this, it) != PackageManager.PERMISSION_GRANTED
                  }
                  
                  if (permissionsToRequest.isNotEmpty()) {
                      Log.d(TAG, "Requesting Deepgram permissions: ${permissionsToRequest.joinToString()}")
                      
                      // Show explanation for critical audio permissions
                      if (permissionsToRequest.any { it in criticalPermissions }) {
                          Toast.makeText(this, 
                              "Audio permissions are required for Deepgram speech recognition in GTLingua", 
                              Toast.LENGTH_LONG).show()
                      }
                      
                      ActivityCompat.requestPermissions(this, permissionsToRequest.toTypedArray(), PERMISSIONS_REQUEST_CODE)
                  } else {
                      Log.d(TAG, "All Deepgram permissions granted")
                      updateStatus("✅ Deepgram speech recognition ready - 36 languages available")
                  }
              }
              
              private fun setupClickListeners() {
                  // Voice Translation - Primary feature leveraging Deepgram
                  voiceTranslateButton.setOnClickListener {
                      launchGTLinguaVoiceMode()
                  }
                  
                  // Text Translation with GTLingua
                  textTranslateButton.setOnClickListener {
                      launchGTLinguaTextMode()
                  }
                  
                  // Camera OCR integration
                  cameraOCRButton.setOnClickListener {
                      launchCameraOCR()
                  }
              }
              
              private fun launchGTLinguaVoiceMode() {
                  if (!checkAudioPermissions()) {
                      Toast.makeText(this, "Microphone permission required for voice translation", Toast.LENGTH_LONG).show()
                      return
                  }
                  
                  val intent = Intent(this, GTLinguaActivity::class.java).apply {
                      putExtra("mode", "voice_primary")
                      putExtra("auto_start_voice", true)
                      putExtra("focus_microphone", true)
                      putExtra("enable_deepgram", true)
                      
                      // Pass any quick text as context
                      val quickText = quickTextInput.text.toString().trim()
                      if (quickText.isNotEmpty()) {
                          putExtra("context_text", quickText)
                      }
                  }
                  
                  startActivityForResult(intent, GTLINGUA_REQUEST_CODE)
                  updateStatus("🎤 Launching GTLingua voice mode with Deepgram...")
              }
              
              private fun launchGTLinguaTextMode() {
                  val intent = Intent(this, GTLinguaActivity::class.java).apply {
                      putExtra("mode", "text_primary")
                      
                      // Pre-fill with quick text if available
                      val quickText = quickTextInput.text.toString().trim()
                      if (quickText.isNotEmpty()) {
                          putExtra("prefill_text", quickText)
                          putExtra("url_params", "?text=${java.net.URLEncoder.encode(quickText, "UTF-8")}")
                      }
                  }
                  
                  startActivityForResult(intent, GTLINGUA_REQUEST_CODE)
                  updateStatus("📝 Launching GTLingua text mode...")
              }
              
              private fun launchCameraOCR() {
                  if (!checkCameraPermissions()) {
                      Toast.makeText(this, "Camera permission required for OCR", Toast.LENGTH_SHORT).show()
                      return
                  }
                  
                  val intent = Intent(this, CameraOCRActivity::class.java)
                  startActivityForResult(intent, CAMERA_REQUEST_CODE)
                  updateStatus("📷 Launching camera OCR...")
              }
              
              private fun checkAudioPermissions(): Boolean {
                  return ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED
              }
              
              private fun checkCameraPermissions(): Boolean {
                  return ContextCompat.checkSelfPermission(this, Manifest.permission.CAMERA) == PackageManager.PERMISSION_GRANTED
              }
              
              private fun handleIncomingIntent() {
                  val intent = intent
                  if (Intent.ACTION_SEND == intent.action && "text/plain" == intent.type) {
                      val sharedText = intent.getStringExtra(Intent.EXTRA_TEXT)
                      if (!sharedText.isNullOrEmpty()) {
                          quickTextInput.setText(sharedText)
                          updateStatus("📤 Text received - ready to translate with GTLingua")
                          
                          // Auto-launch GTLingua with shared text
                          launchGTLinguaTextMode()
                      }
                  }
              }
              
              private fun loadRecentTranslations() {
                  val recent = preferencesManager.getRecentTranslations()
                  if (recent.isNotEmpty()) {
                      recentTranslations.text = "Recent: ${recent.take(3).joinToString(" • ")}"
                  } else {
                      recentTranslations.text = "No recent translations"
                  }
              }
              
              private fun updateStatus(message: String) {
                  statusText.text = message
                  Log.d(TAG, "Status: $message")
              }
              
              override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
                  super.onActivityResult(requestCode, resultCode, data)
                  
                  when (requestCode) {
                      GTLINGUA_REQUEST_CODE -> {
                          if (resultCode == RESULT_OK && data != null) {
                              val translation = data.getStringExtra("translation_result")
                              val sourceText = data.getStringExtra("source_text")
                              
                              if (!translation.isNullOrEmpty()) {
                                  // Save to recent translations
                                  preferencesManager.addRecentTranslation(translation, sourceText)
                                  loadRecentTranslations()
                                  
                                  updateStatus("✅ Translation completed via GTLingua")
                                  
                                  // Clear quick input if it was used
                                  if (!sourceText.isNullOrEmpty() && quickTextInput.text.toString().trim() == sourceText) {
                                      quickTextInput.setText("")
                                  }
                              }
                          }
                      }
                      CAMERA_REQUEST_CODE -> {
                          if (resultCode == RESULT_OK && data != null) {
                              val extractedText = data.getStringExtra("extracted_text")
                              if (!extractedText.isNullOrEmpty()) {
                                  quickTextInput.setText(extractedText)
                                  updateStatus("📷 Text extracted - ready for GTLingua translation")
                                  
                                  // Auto-launch text translation with OCR result
                                  launchGTLinguaTextMode()
                              }
                          }
                      }
                  }
              }
              
              override fun onRequestPermissionsResult(requestCode: Int, permissions: Array<out String>, grantResults: IntArray) {
                  super.onRequestPermissionsResult(requestCode, permissions, grantResults)
                  
                  if (requestCode == PERMISSIONS_REQUEST_CODE) {
                      val audioGranted = permissions.indices.any { 
                          permissions[it] == Manifest.permission.RECORD_AUDIO && 
                          grantResults[it] == PackageManager.PERMISSION_GRANTED 
                      }
                      
                      val audioSettingsGranted = permissions.indices.any { 
                          permissions[it] == Manifest.permission.MODIFY_AUDIO_SETTINGS && 
                          grantResults[it] == PackageManager.PERMISSION_GRANTED 
                      }
                      
                      when {
                          audioGranted && audioSettingsGranted -> {
                              updateStatus("✅ Deepgram speech recognition enabled - Voice translation ready")
                          }
                          audioGranted -> {
                              updateStatus("⚠️ Basic voice enabled - Some Deepgram features may be limited")
                          }
                          else -> {
                              updateStatus("❌ Voice features disabled - Text translation still available")
                              Toast.makeText(this, 
                                  "Voice translation requires microphone access for Deepgram speech recognition", 
                                  Toast.LENGTH_LONG).show()
                          }
                      }
                  }
              }
              
              override fun onResume() {
                  super.onResume()
                  loadRecentTranslations()
              }
          }
          EOF

          import android.app.Activity
          import android.content.Intent
          import android.graphics.Bitmap
          import android.os.Bundle
          import android.util.Log

          import android.view.View

          import android.widget.ProgressBar

          import com.lingualink.gtlingua.R

          class GTLinguaActivity : AppCompatActivity() {
              
              companion object {
                  private const val TAG = "GTLinguaActivity"
                  private const val GTLINGUA_URL = "https://gtlingua.com"
                  private const val FALLBACK_URL = "https://translate.google.com"
              }
              
              private lateinit var webView: WebView
              private lateinit var progressBar: ProgressBar
              private lateinit var toolbar: Toolbar
              private lateinit var bridge: GTLinguaBridge
              
              private var translationMode: String = "text_primary"
              private var autoStartVoice: Boolean = false
              private var prefillText: String? = null
              
              @SuppressLint("SetJavaScriptEnabled")
              override fun onCreate(savedInstanceState: Bundle?) {
                  super.onCreate(savedInstanceState)
                  setContentView(R.layout.activity_gtlingua)
                  
                  // Get launch parameters
                  translationMode = intent.getStringExtra("mode") ?: "text_primary"
                  autoStartVoice = intent.getBooleanExtra("auto_start_voice", false)
                  prefillText = intent.getStringExtra("prefill_text")
                  
                  initializeComponents()
                  setupToolbar()
                  setupWebView()
                  
                  loadGTLingua()
                  
                  Log.d(TAG, "GTLingua WebView launched in mode: $translationMode")
              }
              
              private fun initializeComponents() {
                  webView = findViewById(R.id.gtlinguaWebView)
                  progressBar = findViewById(R.id.gtlinguaProgressBar)
                  toolbar = findViewById(R.id.gtlinguaToolbar)
                  
                  bridge = GTLinguaBridge(this)
              }
              
              private fun setupToolbar() {
                  setSupportActionBar(toolbar)
                  supportActionBar?.apply {
                      setDisplayHomeAsUpEnabled(true)
                      setDisplayShowHomeEnabled(true)
                      title = when (translationMode) {
                          "voice_primary" -> "🎤 GTLingua Voice (Deepgram)"
                          "text_primary" -> "📝 GTLingua Translator"
                          else -> "GTLingua AI Translator"
                      }
                  }
              }
              
              @SuppressLint("SetJavaScriptEnabled")
              private fun setupWebView() {
                  val webSettings = webView.settings
                  
                  // OPTIMIZED SETTINGS FOR GTLINGUA + DEEPGRAM
                  webSettings.javaScriptEnabled = true
                  webSettings.mediaPlaybackRequiresUserGesture = false  // Critical for Deepgram
                  webSettings.domStorageEnabled = true
                  webSettings.databaseEnabled = true
                  webSettings.cacheMode = WebSettings.LOAD_DEFAULT
                  
                  // AUDIO/MICROPHONE SETTINGS FOR DEEPGRAM
                  webSettings.javaScriptCanOpenWindowsAutomatically = true
                  webSettings.allowContentAccess = true
                  webSettings.allowFileAccess = true
                  webSettings.allowFileAccessFromFileURLs = false

          package com.lingualink.gtlingua.utils
          
          import android.content.Context
          import android.content.SharedPreferences
          import org.json.JSONArray
          import org.json.JSONObject
          
          class PreferencesManager(context: Context) {
              
              companion object {
                  private const val PREFS_NAME = "gtlingua_prefs"
                  private const val KEY_RECENT_TRANSLATIONS = "recent_translations"
                  private const val KEY_PREFERRED_VOICE_LANG = "preferred_voice_lang"
                  private const val KEY_AUTO_START_VOICE = "auto_start_voice"
                  private const val KEY_DEEPGRAM_ENABLED = "deepgram_enabled"
                  private const val MAX_RECENT_TRANSLATIONS = 10
              }
              
              private val prefs: SharedPreferences = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
              
              fun addRecentTranslation(translation: String, sourceText: String?) {
                  val recent = getRecentTranslationsJson()
                  
                  val newTranslation = JSONObject().apply {
                      put("translation", translation)
                      put("source", sourceText ?: "")
                      put("timestamp", System.currentTimeMillis())
                  }
                  
                  recent.put(0, newTranslation)
                  
                  // Keep only the most recent translations
                  while (recent.length() > MAX_RECENT_TRANSLATIONS) {
                      recent.remove(recent.length() - 1)
                  }
                  
                  prefs.edit()
                      .putString(KEY_RECENT_TRANSLATIONS, recent.toString())
                      .apply()
              }
              
              fun getRecentTranslations(): List<String> {
                  val recent = getRecentTranslationsJson()
                  val translations = mutableListOf<String>()
                  
                  for (i in 0 until recent.length()) {
                      val item = recent.optJSONObject(i)
                      item?.optString("translation")?.let { translations.add(it) }
                  }
                  
                  return translations
              }
              
              private fun getRecentTranslationsJson(): JSONArray {
                  val recentString = prefs.getString(KEY_RECENT_TRANSLATIONS, "[]")
                  return try {
                      JSONArray(recentString)
                  } catch (e: Exception) {
                      JSONArray()
                  }
              }
              
              fun setPreferredVoiceLanguage(language: String) {
                  prefs.edit().putString(KEY_PREFERRED_VOICE_LANG, language).apply()
              }
              
              fun getPreferredVoiceLanguage(): String {
                  return prefs.getString(KEY_PREFERRED_VOICE_LANG, "en") ?: "en"
              }
              
              fun setAutoStartVoice(enabled: Boolean) {
                  prefs.edit().putBoolean(KEY_AUTO_START_VOICE, enabled).apply()
              }
              
              fun getAutoStartVoice(): Boolean {
                  return prefs.getBoolean(KEY_AUTO_START_VOICE, false)
              }
              
              fun setDeepgramEnabled(enabled: Boolean) {
                  prefs.edit().putBoolean(KEY_DEEPGRAM_ENABLED, enabled).apply()
              }
              
              fun isDeepgramEnabled(): Boolean {
                  return prefs.getBoolean(KEY_DEEPGRAM_ENABLED, true)
              }
          }
